! Subroutine:  read_kamd_namelist
!
! Description: Read the optional &kamd namelist.  Defaults are set in
!              init_mdin_ctrl_dat before this is called; absent namelist =>
!              KAMD stays disabled (pseudobd_cycles = 0).  Performs only the
!              cheap, prmtop-independent validation here; mask resolution and
!              atom-list construction happen later in resolve_kamd_masks.
!
!*******************************************************************************
subroutine read_kamd_namelist()

  use file_io_mod
  use gbl_constants_mod, only : error_hdr
  use parallel_dat_mod
  use pmemd_lib_mod

  implicit none

  integer :: ifind
  integer :: g

  namelist /kamd/ pseudobd_cycles, nzero_crossings, &
                  shell_cut, surface_cut, membrane_cut, &
                  kamd_shell_mode, kamd_shell_relax_steps, &
                  kamd_shell_max_angle, kamd_shell_area_floor, kamd_shell_area_reference, &
                  refer_link, region_type, &
                  refer_mask, solvent_mask, membrane_mask, &
                  membrane_axis, &
                  ion_mask, ion_traj_outfreq, ion_traj_file

  rewind(mdin)
  call nmlsrc('kamd', mdin, ifind)
  if (ifind == 0) return            ! KAMD not requested; defaults stand

  read(mdin, nml = kamd)

  if (kamd_shell_mode < 0 .or. kamd_shell_mode > 1) then
    write(mdout, '(a,a)') error_hdr, '&kamd: kamd_shell_mode must be 0 or 1'
    call mexit(6, 1)
  end if
  if (kamd_shell_relax_steps < 0 .or. kamd_shell_max_angle <= 0.0d0 .or. &
      kamd_shell_max_angle > 180.0d0 .or. kamd_shell_area_floor < 0.0d0 .or. &
      kamd_shell_area_floor >= 1.0d0 .or. kamd_shell_area_reference < 0.0d0) then
    write(mdout, '(a,a)') error_hdr, '&kamd: invalid membrane shell relaxation/angle/area setting'
    call mexit(6, 1)
  end if

  ! Ion-trajectory validation (independent of pseudobd_cycles).
  if (ion_traj_outfreq < 0) then
    write(mdout, '(a,a)') error_hdr, '&kamd: ion_traj_outfreq must be >= 0'
    call mexit(6, 1)
  end if
  if (ion_traj_outfreq > 0 .and. len_trim(ion_mask) == 0) then
    write(mdout, '(a,a)') error_hdr, '&kamd: ion_mask is required when ion_traj_outfreq > 0'
    call mexit(6, 1)
  end if

#if !defined(CUDA) || defined(MPI)
  ! The KAMD code in runmd is CUDA-only and has been validated on a single GPU only.
  ! Refuse instead of silently running plain MD (CPU pmemd) or an untested multi-GPU path.
  if (pseudobd_cycles > 0 .or. ion_traj_outfreq > 0) then
    write(mdout, '(a,a)') error_hdr, &
      '&kamd: KAMD is available only in single-GPU pmemd.cuda (pmemd.cuda_SPFP / pmemd.cuda_DPFP)'
    call mexit(6, 1)
  end if
#endif

  if (pseudobd_cycles <= 0) return  ! Namelist present but disabled; nothing to validate

  if (nzero_crossings <= 0) then
    write(mdout, '(a,a)') error_hdr, '&kamd: nzero_crossings must be > 0'
    call mexit(6, 1)
  end if
  if (shell_cut <= 0.0d0 .or. surface_cut <= 0.0d0) then
    write(mdout, '(a,a)') error_hdr, '&kamd: shell_cut and surface_cut must be > 0'
    call mexit(6, 1)
  end if
  if (surface_cut > shell_cut) then
    write(mdout, '(a,a)') error_hdr, '&kamd: surface_cut must be <= shell_cut'
    call mexit(6, 1)
  end if
  if (len_trim(solvent_mask) == 0) then
    write(mdout, '(a,a)') error_hdr, '&kamd: solvent_mask is required when pseudobd_cycles > 0'
    call mexit(6, 1)
  end if

  ! Count active groups and validate refer_link entries.
  kamd_ngroups = 0
  do g = 1, KAMD_MAX_GROUPS
    if (len_trim(refer_mask(g)) > 0) kamd_ngroups = kamd_ngroups + 1
  end do
  if (kamd_ngroups == 0) then
    write(mdout, '(a,a)') error_hdr, '&kamd: at least one refer_mask(i) must be set'
    call mexit(6, 1)
  end if

  ! refer_mask entries must be packed into 1..kamd_ngroups (no gaps), so the
  ! group index seen on the GPU side matches the position in the user input.
  do g = 1, kamd_ngroups
    if (len_trim(refer_mask(g)) == 0) then
      write(mdout, '(a,a,i0,a)') error_hdr, &
          '&kamd: refer_mask entries must be packed; refer_mask(', g, ') is empty'
      call mexit(6, 1)
    end if
  end do
  do g = kamd_ngroups + 1, KAMD_MAX_GROUPS
    if (len_trim(refer_mask(g)) > 0) then
      write(mdout, '(a,a)') error_hdr, &
          '&kamd: refer_mask entries must be contiguous (no gaps)'
      call mexit(6, 1)
    end if
  end do

  do g = 1, kamd_ngroups
    if (refer_link(g) < 0 .or. refer_link(g) > kamd_ngroups) then
      write(mdout, '(a,a,i0,a,i0,a)') error_hdr, '&kamd: refer_link(', g, &
          ') = ', refer_link(g), ' is out of range [0, kamd_ngroups]'
      call mexit(6, 1)
    end if
    if (refer_link(g) == g) then
      write(mdout, '(a,a,i0,a)') error_hdr, '&kamd: refer_link(', g, &
          ') cannot reference its own group'
      call mexit(6, 1)
    end if
  end do

  ! Validate region_type and membrane setup.  region_type defaults to 1
  ! (solution) for every group; transmembrane groups (type 2) require the
  ! membrane bath to be specified.
  do g = 1, kamd_ngroups
    if (region_type(g) /= KAMD_RT_SOLUTION .and. region_type(g) /= KAMD_RT_TRANSMEMBRANE) then
      write(mdout, '(a,a,i0,a,i0,a)') error_hdr, '&kamd: region_type(', g, &
          ') = ', region_type(g), ' is invalid (1=solution, 2=transmembrane)'
      call mexit(6, 1)
    end if
  end do
  if (any(region_type(1:kamd_ngroups) == KAMD_RT_TRANSMEMBRANE)) then
    if (len_trim(membrane_mask) == 0) then
      write(mdout, '(a,a)') error_hdr, &
          '&kamd: membrane_mask is required when any region_type == 2'
      call mexit(6, 1)
    end if
    if (membrane_axis < 1 .or. membrane_axis > 3) then
      write(mdout, '(a,a,i0,a)') error_hdr, &
          '&kamd: membrane_axis = ', membrane_axis, ' is invalid (1=x, 2=y, 3=z)'
      call mexit(6, 1)
    end if
    if (membrane_cut <= 0.0d0) then
      write(mdout, '(a,a)') error_hdr, '&kamd: membrane_cut must be > 0'
      call mexit(6, 1)
    end if
  end if
  if (kamd_shell_mode == 1 .and. &
      .not. any(region_type(1:kamd_ngroups) == KAMD_RT_TRANSMEMBRANE)) then
    write(mdout, '(a,a)') error_hdr, '&kamd: kamd_shell_mode requires a transmembrane region'
    call mexit(6, 1)
  end if

end subroutine read_kamd_namelist

!*******************************************************************************
!
! Subroutine:  resolve_kamd_masks
!
! Description: Evaluate the &kamd refer_mask(:) and solvent_mask strings into
!              concrete atom-index lists, using pmemd's existing AmberMask
!              parser (findmask_mod::atommask).  Called once from master_setup
!              after the prmtop and coordinates have been loaded.
!
!              Reference groups may resolve to discontiguous atom sets.
!              The solvent mask must currently resolve to a single contiguous
!              atom range (the GPU first-shell kernel uses a [start,end] range
!              filter); we error out cleanly if it does not.
!
!*******************************************************************************
subroutine resolve_kamd_masks(natom, nres, igraph, isymbl, ipres, lbres, crd)

  use findmask_mod, only : atommask
  use gbl_constants_mod, only : error_hdr
  use parallel_dat_mod
  use file_io_mod
  use pmemd_lib_mod

  implicit none

  integer,           intent(in) :: natom, nres
  integer,           intent(in) :: ipres(nres)
  character(len=4),  intent(in) :: igraph(natom), isymbl(natom), lbres(nres)
  double precision,  intent(in) :: crd(3*natom)

  integer, allocatable :: scratch_mask(:)
  integer, allocatable :: owner_mask(:)   ! per-atom claim tag for overlap detection
  integer              :: g, a, idx, count_g
  character(len=1024)  :: mask_buf

  if (pseudobd_cycles <= 0 .and. ion_traj_outfreq <= 0) return  ! KAMD disabled: nothing to do

  allocate(scratch_mask(natom))
  allocate(owner_mask(natom))

  ! Only resolve reference groups and solvent when pseudobd is active
  if (pseudobd_cycles > 0) then
    allocate(kamd_refer_count(kamd_ngroups))
    allocate(kamd_refer_offset(kamd_ngroups))

  ! Atom-claim tracker for overlap detection.
  !   0           = unclaimed
  !   1..ngroups  = claimed by refer_mask(g)
  !   -1          = claimed by solvent_mask
  ! Any double-claim is a hard error: a single atom cannot belong to two
  ! reference groups, nor to both a reference group and the solvent bath.
  owner_mask = 0

  ! First pass: count atoms per group, sniff for refer/refer overlap.
  kamd_total_ref_atoms = 0
  do g = 1, kamd_ngroups
    mask_buf = refer_mask(g)
    call atommask(natom, nres, 0, igraph, isymbl, ipres, lbres, crd, mask_buf, scratch_mask)
    count_g = sum(scratch_mask)
    if (count_g <= 0) then
      write(mdout, '(a,a,i0,a,a,a)') error_hdr, '&kamd: refer_mask(', g, &
          ') = "', trim(refer_mask(g)), '" matched no atoms'
      call mexit(6, 1)
    end if
    do a = 1, natom
      if (scratch_mask(a) /= 0) then
        if (owner_mask(a) > 0) then
          write(mdout, '(a,a,i0,a,i0,a,i0,a)') error_hdr, &
              '&kamd: atom ', a, ' is matched by both refer_mask(', &
              owner_mask(a), ') and refer_mask(', g, '). Reference groups must not overlap.'
          call mexit(6, 1)
        end if
        owner_mask(a) = g
      end if
    end do
    kamd_refer_count(g)  = count_g
    kamd_refer_offset(g) = kamd_total_ref_atoms
    kamd_total_ref_atoms = kamd_total_ref_atoms + count_g
  end do

  ! Second pass: fill the flat (atom_idx, group_idx) lists.
  allocate(kamd_refer_atoms(kamd_total_ref_atoms))
  allocate(kamd_refer_groups(kamd_total_ref_atoms))
  do g = 1, kamd_ngroups
    mask_buf = refer_mask(g)
    call atommask(natom, nres, 0, igraph, isymbl, ipres, lbres, crd, mask_buf, scratch_mask)
    idx = kamd_refer_offset(g)
    do a = 1, natom
      if (scratch_mask(a) /= 0) then
        idx = idx + 1
        kamd_refer_atoms(idx)  = a
        kamd_refer_groups(idx) = g
      end if
    end do
    write(mdout, '(a,i0,a,a,a,i0,a)') '     KAMD refer_mask(', g, ') = "', &
        trim(refer_mask(g)), '" matches ', kamd_refer_count(g), ' atoms'
  end do

  ! Solvent mask: must resolve to a contiguous range for the current GPU
  ! kernel.  We accept any mask whose selected atoms span exactly one
  ! contiguous block of indices.
  mask_buf = solvent_mask
  call atommask(natom, nres, 0, igraph, isymbl, ipres, lbres, crd, mask_buf, scratch_mask)
  kamd_solvent_start = 0
  kamd_solvent_end   = 0
  do a = 1, natom
    if (scratch_mask(a) /= 0) then
      if (kamd_solvent_start == 0) kamd_solvent_start = a
      kamd_solvent_end = a
    end if
  end do
  if (kamd_solvent_start == 0) then
    write(mdout, '(a,a,a,a)') error_hdr, '&kamd: solvent_mask = "', &
        trim(solvent_mask), '" matched no atoms'
    call mexit(6, 1)
  end if
  if (sum(scratch_mask) /= (kamd_solvent_end - kamd_solvent_start + 1)) then
    write(mdout, '(a,a,a,a)') error_hdr, '&kamd: solvent_mask = "', &
        trim(solvent_mask), '" must resolve to a contiguous atom range'
    write(mdout, '(a,i0,a,i0,a,i0,a)') &
        '     resolved span = [', kamd_solvent_start, ', ', kamd_solvent_end, &
        '] but only ', sum(scratch_mask), ' atoms selected (gaps present).'
    write(mdout, '(a)') &
        '     Reorder solvent residues to be contiguous in the prmtop, or split into a separate run.'
    call mexit(6, 1)
  end if

  ! refer/solvent overlap check: any atom claimed by a refer_mask cannot also
  ! be in the solvent bath.
  do a = kamd_solvent_start, kamd_solvent_end
    if (scratch_mask(a) /= 0 .and. owner_mask(a) > 0) then
      write(mdout, '(a,a,i0,a,i0,a)') error_hdr, &
          '&kamd: atom ', a, ' is matched by both refer_mask(', &
          owner_mask(a), ') and solvent_mask. Reference groups and the solvent bath must not overlap.'
      call mexit(6, 1)
    end if
    ! Mark solvent atoms with sentinel -1 so the membrane pass below can detect
    ! solvent/membrane overlap as well.
    if (scratch_mask(a) /= 0) owner_mask(a) = -1
  end do

  write(mdout, '(a,a,a,i0,a,i0,a)') '     KAMD solvent_mask = "', &
      trim(solvent_mask), '" -> contiguous range [', kamd_solvent_start, &
      ', ', kamd_solvent_end, ']'

    ! Membrane mask: required iff any region is transmembrane.  Same contiguous-
  ! range constraint as solvent (kernel uses [start,end] filter), and no atom
  ! may be claimed by both the membrane bath and any reference group or the
  ! solvent bath.
  if (any(region_type(1:kamd_ngroups) == KAMD_RT_TRANSMEMBRANE)) then
    mask_buf = membrane_mask
    call atommask(natom, nres, 0, igraph, isymbl, ipres, lbres, crd, mask_buf, scratch_mask)
    kamd_membrane_start = 0
    kamd_membrane_end   = 0
    do a = 1, natom
      if (scratch_mask(a) /= 0) then
        if (kamd_membrane_start == 0) kamd_membrane_start = a
        kamd_membrane_end = a
      end if
    end do
    if (kamd_membrane_start == 0) then
      write(mdout, '(a,a,a,a)') error_hdr, '&kamd: membrane_mask = "', &
          trim(membrane_mask), '" matched no atoms'
      call mexit(6, 1)
    end if
    if (sum(scratch_mask) /= (kamd_membrane_end - kamd_membrane_start + 1)) then
      write(mdout, '(a,a,a,a)') error_hdr, '&kamd: membrane_mask = "', &
          trim(membrane_mask), '" must resolve to a contiguous atom range'
      write(mdout, '(a,i0,a,i0,a,i0,a)') &
          '     resolved span = [', kamd_membrane_start, ', ', kamd_membrane_end, &
          '] but only ', sum(scratch_mask), ' atoms selected (gaps present).'
      call mexit(6, 1)
    end if
    do a = kamd_membrane_start, kamd_membrane_end
      if (scratch_mask(a) /= 0 .and. owner_mask(a) > 0) then
        write(mdout, '(a,a,i0,a,i0,a)') error_hdr, &
            '&kamd: atom ', a, ' is matched by both refer_mask(', &
            owner_mask(a), ') and membrane_mask. Reference groups and the membrane bath must not overlap.'
        call mexit(6, 1)
      end if
      if (scratch_mask(a) /= 0 .and. owner_mask(a) == -1) then
        write(mdout, '(a,a,i0,a)') error_hdr, &
            '&kamd: atom ', a, ' is matched by both solvent_mask and membrane_mask. The two baths must be disjoint.'
        call mexit(6, 1)
      end if
    end do
    write(mdout, '(a,a,a,i0,a,i0,a)') '     KAMD membrane_mask = "', &
        trim(membrane_mask), '" -> contiguous range [', kamd_membrane_start, &
        ', ', kamd_membrane_end, ']'
  end if

  end if

  ! ----- Ion-trajectory setup (independent of pseudobd_cycles) -----
  if (ion_traj_outfreq > 0) then

    ! Resolve ion_mask -> kamd_ion_atoms (1-based).
    mask_buf = ion_mask
    call atommask(natom, nres, 0, igraph, isymbl, ipres, lbres, crd, mask_buf, scratch_mask)
    kamd_nions = sum(scratch_mask)
    if (kamd_nions <= 0) then
      write(mdout, '(a,a,a,a)') error_hdr, '&kamd: ion_mask = "', trim(ion_mask), '" matched no atoms'
      call mexit(6, 1)
    end if
    allocate(kamd_ion_atoms(kamd_nions))
    allocate(kamd_ion_xyz(3, kamd_nions))
    idx = 0
    do a = 1, natom
      if (scratch_mask(a) /= 0) then
        idx = idx + 1
        kamd_ion_atoms(idx) = a
      end if
    end do
    write(mdout, '(a,a,a,i0,a)') '     KAMD ion_mask = "', trim(ion_mask), '" matches ', kamd_nions, ' atoms'

    ! Membrane COM reference is required for the ion-trajectory file.  If the TM
    ! block above did not already resolve it, resolve membrane_mask here with the
    ! same contiguous-range requirement (the COM reduction iterates [start,end]).
    if (kamd_membrane_start == 0) then
      mask_buf = membrane_mask
      call atommask(natom, nres, 0, igraph, isymbl, ipres, lbres, crd, mask_buf, scratch_mask)
      kamd_membrane_start = 0
      kamd_membrane_end   = 0
      do a = 1, natom
        if (scratch_mask(a) /= 0) then
          if (kamd_membrane_start == 0) kamd_membrane_start = a
          kamd_membrane_end = a
        end if
      end do
      if (kamd_membrane_start == 0) then
        write(mdout, '(a,a)') error_hdr, '&kamd: membrane_mask is required for ion trajectory output (membrane COM)'
        call mexit(6, 1)
      end if
      if (sum(scratch_mask) /= (kamd_membrane_end - kamd_membrane_start + 1)) then
        write(mdout, '(a,a,a,a)') error_hdr, '&kamd: membrane_mask = "', trim(membrane_mask), '" must resolve to a contiguous atom range'
        call mexit(6, 1)
      end if
      write(mdout, '(a,a,a,i0,a,i0,a)') '     KAMD membrane_mask (ion-traj COM) -> contiguous range [', &
          kamd_membrane_start, ', ', kamd_membrane_end, ']'
    end if

  end if

  deallocate(scratch_mask)
  deallocate(owner_mask)

end subroutine resolve_kamd_masks
